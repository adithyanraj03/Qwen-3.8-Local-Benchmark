daily_t = df.dropna(subset=["temperature_c"]).groupby("date").agg(
    temp=("temperature_c", "mean"), rides=("ride_id", "size"))
r = daily_t.temp.corr(daily_t.rides)
slope = np.polyfit(daily_t.temp, daily_t.rides, 1)
fig, ax = plt.subplots(figsize=(9, 4.8))
ax.scatter(daily_t.temp, daily_t.rides, s=22, alpha=0.6, color=BLUE)
xs = np.linspace(daily_t.temp.min(), daily_t.temp.max(), 50)
ax.plot(xs, np.polyval(slope, xs), color=RED, lw=2,
        label=f"fit: +{slope[0]:.0f} rides per +1 °C (r = {r:.2f})")
ax.set_xlabel("Mean daily temperature (°C)"); ax.set_ylabel("Rides that day")
ax.legend(frameon=False); ax.set_title("Daily ride volume vs mean temperature (365 days)")
fig.tight_layout(); fig.savefig(CHARTS / "temperature_vs_volume.png", dpi=130); plt.show()
buckets = pd.cut(daily_t.temp, [-10, 0, 10, 20, 25, 30])
print("mean rides/day by temperature band:")
print(daily_t.groupby(buckets, observed=True).rides.mean().round(0).to_string())
print(f"Pearson r = {r:.3f} across days")
cold = daily_t[daily_t.temp < 0].rides.mean(); mild = daily_t[daily_t.temp.between(15,25)].rides.mean()
print(f"days below 0 °C average {cold:.0f} rides vs {mild:.0f} on 15–25 °C days "
      f"({cold/mild:.0%})")

    rides=("ride_id", "size"), median_min=("duration_min", "median"),
    mean_km=("distance_km", "mean"), mean_fare=("fare", "mean"),
    revenue=("fare", "sum"), weekend_share=("is_weekend", "mean"),
